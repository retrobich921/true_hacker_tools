def compute_pi(s):
    n=len(s)
    p=[0]*n
    for i in range(1,n):
        j=p[i-1]
        while j>0 and s[i]!=s[j]:
            j=p[j-1]
        if s[i]==s[j]:
            j+=1
        p[i]=j
    return p
def compute_z(s):
    n=len(s)
    z=[0]*n
    l=0
    r=0
    for i in range(1,n):
        if i<=r:
            z[i]=min(r-i+1,z[i-l])
        while i+z[i]<n and s[z[i]]==s[i+z[i]]:
            z[i]+=1
        if i+z[i]-1>r:
            l=i
            r=i+z[i]-1
    return z
import sys
sys.setrecursionlimit(300000)
class Node:
    __slots__=['c','cnt']
    def __init__(self):
        self.c={}
        self.cnt=0
class Trie:
    def __init__(self):
        self.rt=Node()
    def insert(self,s):
        curr=self.rt
        for x in s:
            if x not in curr.c:
                curr.c[x]=Node()
            curr=curr.c[x]
            curr.cnt+=1
    def count(self,s):
        curr=self.rt
        for x in s:
            if x not in curr.c:
                return 0
            curr=curr.c[x]
        return curr.cnt
